export const vnpayReturn = async (req: Request, res: Response, next: NextFunction) => {
  const query = req.query as Record<string, string>;

  if (!paymentService.verifyVnpayCallback(query)) {
    return res.redirect(paymentService.buildPaymentResultRedirect(query.vnp_TxnRef ?? "", "invalid"));
  }

  const success = query.vnp_ResponseCode === "00";

  try {
    await paymentService.applyGatewayResult({
      gateway: "VNPAY",
      transactionCode: query.vnp_TransactionNo ?? query.vnp_TxnRef,
      orderNumber: query.vnp_TxnRef,
      amount: Number(query.vnp_Amount) / 100,
      success,
      message: query.vnp_Message,
      rawPayload: query,
    });
  } catch (error) {
    return res.redirect(paymentService.buildPaymentResultRedirect(query.vnp_TxnRef, "invalid"));
  }

  return res.redirect(
    paymentService.buildPaymentResultRedirect(query.vnp_TxnRef, success ? "success" : "failed")
  );
};

import { Request, Response, NextFunction } from "express";
import { verifyToken } from "../utils/jwt";

/**
 * Má»Ÿ rá»™ng Request cá»§a Express
 */
declare global {
  namespace Express {
    interface Request {
      user?: {
        userId: string;
        role: "customer" | "admin";
      };
    }
  }
}

/**
 * Middleware xĂ¡c thá»±c Ä‘Äƒng nháº­p
 */
const authMiddleware = (
  req: Request,
  res: Response,
  next: NextFunction
): void => {
  try {
    // Láº¥y Authorization Header
    const authHeader = req.headers.authorization;

    if (!authHeader) {
      res.status(401).json({
        success: false,
        message: "Báº¡n chÆ°a Ä‘Äƒng nháº­p",
      });

      return;
    }

    // Authorization pháº£i cĂ³ dáº¡ng:
    // Bearer token
    if (!authHeader.startsWith("Bearer ")) {
      res.status(401).json({
        success: false,
        message: "Authorization khĂ´ng há»£p lá»‡",
      });

      return;
    }

    // Láº¥y token
    const token = authHeader.split(" ")[1];

    if (!token) {
      res.status(401).json({
        success: false,
        message: "Token khĂ´ng tá»“n táº¡i",
      });

      return;
    }

    // Verify JWT
    const decoded = verifyToken(token);

    // Gáº¯n user vĂ o request
    req.user = {
      userId: decoded.userId,
      role: decoded.role,
    };

    // Cho request Ä‘i tiáº¿p
    next();
  } catch (error) {
    res.status(401).json({
      success: false,
      message: "Token khĂ´ng há»£p lá»‡ hoáº·c Ä‘Ă£ háº¿t háº¡n",
    });
  }
};

export default authMiddleware;

 * - Äiá»ƒm xá»­ lĂ½ lá»—i DUY NHáº¤T cá»§a toĂ n bá»™ API.

 * - Map cĂ¡c lá»—i Mongoose phá»• biáº¿n (ValidationError, CastError, trĂ¹ng unique)
 *   vá» 400 vá»›i thĂ´ng bĂ¡o thĂ¢n thiá»‡n.

import type {
  Request,
  Response,
  NextFunction,
} from "express";

interface AppErrorLike extends Error {
  statusCode?: number;
  code?: number;
  keyValue?: Record<string, unknown>;
}

const errorMiddleware = (
  err: AppErrorLike,
  _req: Request,
  res: Response,
  next: NextFunction
): void => {
  // Náº¿u response Ä‘Ă£ Ä‘Æ°á»£c gá»­i, chuyá»ƒn lá»—i cho Express xá»­ lĂ½.
  if (res.headersSent) {
    next(err);
    return;
  }

  let statusCode =
    typeof err.statusCode === "number" &&
    err.statusCode >= 400 &&
    err.statusCode <= 599
      ? err.statusCode
      : 500;

  let message = err.message || "YĂªu cáº§u khĂ´ng há»£p lá»‡.";

  // Mongoose: validate schema tháº¥t báº¡i (thiáº¿u trÆ°á»ng, sai enum...).
  if (err.name === "ValidationError") {
    statusCode = 400;
  }

  // Mongoose: ObjectId sai Ä‘á»‹nh dáº¡ng.
  if (err.name === "CastError") {
    statusCode = 400;
    message = "ID khĂ´ng Ä‘Ăºng Ä‘á»‹nh dáº¡ng.";
  }

  // Mongo: trĂ¹ng khĂ³a unique (E11000).
  if (err.code === 11000) {
    statusCode = 400;
    const field = err.keyValue ? Object.keys(err.keyValue)[0] : null;
    message = field
      ? `GiĂ¡ trá»‹ cá»§a trÆ°á»ng '${field}' Ä‘Ă£ tá»“n táº¡i.`
      : "Dá»¯ liá»‡u Ä‘Ă£ tá»“n táº¡i (trĂ¹ng khĂ³a duy nháº¥t).";
  }

  // Chá»‰ ghi log chi tiáº¿t phĂ­a server Ä‘á»‘i vá»›i lá»—i 500.
  if (statusCode >= 500) {
    console.error("Internal Server Error:", err);
  }

  const responseMessage =
    statusCode === 500
      ? "ÄĂ£ xáº£y ra lá»—i mĂ¡y chá»§. Vui lĂ²ng thá»­ láº¡i sau."
      : message;

  res.status(statusCode).json({

  });

const resolveShippingInfo = async (
  userId: string,
  payload: CheckoutPayload
): Promise<CheckoutSummaryResponse["shippingAddress"]> => {
  if (payload.addressId) {
    const address = await UserAddress.findOne({
      _id: new Types.ObjectId(payload.addressId),
      userId: new Types.ObjectId(userId),
    }).lean();

    if (!address) {
      throw new AppError(404, "Äá»‹a chá»‰ giao hĂ ng khĂ´ng tá»“n táº¡i hoáº·c khĂ´ng thuá»™c vá» báº¡n.");
    }

    return {

      customerPhone: address.phone,
      addressLine: address.addressLine,
      ward: address.ward ?? null,
      district: address.district ?? null,
      province: address.province ?? null,
    };
  }

  if (!payload.customerName || !payload.customerPhone || !payload.shippingAddress) {
    throw new AppError(400, 
      "Cáº§n cung cáº¥p addressId hoáº·c Ä‘áº§y Ä‘á»§ customerName, customerPhone vĂ  shippingAddress."
    );
  }

  return {

    district: payload.shippingDistrict ?? null,
    province: payload.shippingProvince ?? null,
  };
};

const buildCheckoutSummary = async (
  userId: string,
  payload: CheckoutPayload
): Promise<CheckoutSummaryResponse> => {
  const shipping = await resolveShippingInfo(userId, payload);

  const cartFilter: Record<string, unknown> = {
    userId: new Types.ObjectId(userId),
  };

  if (payload.selectedCartItemIds && payload.selectedCartItemIds.length > 0) {
    cartFilter._id = {
      $in: payload.selectedCartItemIds.map((id) => new Types.ObjectId(id)),
    };
  }

  const cartRecords = await Cart.find(cartFilter)
    .populate({
      path: "productId",
      select: "name sku basePrice salePrice isActive",
    })
    .populate({
      path: "variantId",
      select: "sku price stockQuantity isActive",
    })
    .lean();

  if (cartRecords.length === 0) {
    throw new AppError(400, "Giá» hĂ ng trá»‘ng hoáº·c khĂ´ng cĂ³ sáº£n pháº©m há»£p lá»‡ Ä‘á»ƒ Ä‘áº·t hĂ ng.");
  }

  const items: CheckoutLineItem[] = [];
  let subtotal = 0;

  for (const record of cartRecords) {
    const product = record.productId as any;
    const variant = record.variantId as any;

    if (!product || !product.isActive) {
      throw new AppError(400, 
        "CĂ³ sáº£n pháº©m trong giá» Ä‘Ă£ ngá»«ng kinh doanh, vui lĂ²ng kiá»ƒm tra láº¡i giá» hĂ ng."
      );
    }

    const unitPrice = variant
      ? variant.price
      : product.salePrice ?? product.basePrice;
    const totalPrice = unitPrice * record.quantity;

    if (variant) {
      if (!variant.isActive) {
        throw new AppError(400, `Biáº¿n thá»ƒ cá»§a sáº£n pháº©m ${product.name} Ä‘Ă£ ngá»«ng kinh doanh.`);
      }
      if (variant.stockQuantity < record.quantity) {
        throw new AppError(400, 
          `Sáº£n pháº©m ${product.name} chá»‰ cĂ²n ${variant.stockQuantity} sáº£n pháº©m trong kho.`
        );
      }
    }

    items.push({
      cartItemId: record._id.toString(),
      productId: product._id.toString(),
      variantId: variant ? variant._id.toString() : null,
      productName: product.name,
      productSku: product.sku ?? null,
      variantSku: variant ? variant.sku : null,
      unitPrice,
      quantity: record.quantity,
      totalPrice,
    });

    subtotal += totalPrice;
  }

  let discountAmount = 0;
  let couponRef: CheckoutSummaryResponse["coupon"] = null;

  if (payload.couponCode) {
    const couponResult = await validateAndCalculateCoupon(
      userId,
      payload.couponCode,
      subtotal
    );
    discountAmount = couponResult.discountAmount;
    couponRef = {
      id: couponResult.coupon.id,
      code: couponResult.coupon.code,
      name: couponResult.coupon.name,
    };
  }

  const shippingFee = payload.shippingFee ?? 0;
  const taxAmount = 0;
  const totalAmount = Math.max(
    0,
    subtotal - discountAmount + shippingFee + taxAmount
  );

  return {
    items,
    subtotal,
    discountAmount,
    shippingFee,
    taxAmount,
    totalAmount,
    coupon: couponRef,
    shippingAddress: shipping,
  };
};

const generateOrderNumber = async (): Promise<string> => {
  const now = new Date();
  const datePart = [
    now.getFullYear(),
    String(now.getMonth() + 1).padStart(2, "0"),
    String(now.getDate()).padStart(2, "0"),
  ].join("");

  for (let attempt = 0; attempt < 5; attempt += 1) {
    const randomPart = Math.floor(100000 + Math.random() * 900000);
    const candidate = `DH-${datePart}-${randomPart}`;
    const existed = await Order.exists({ orderNumber: candidate });
    if (!existed) {
      return candidate;
    }
  }

  return `DH-${datePart}-${Date.now()}`;
};

export const placeOrder = async (
  userId: string,
  payload: CheckoutPayload
): Promise<{ order: IOrder; items: IOrderItem[] }> => {
  if (!payload.paymentMethod) {
    throw new AppError(400, "PhÆ°Æ¡ng thá»©c thanh toĂ¡n lĂ  báº¯t buá»™c.");
  }

  const user = await User.findById(new Types.ObjectId(userId)).select("email");
  if (!user) {
    throw new AppError(404, "TĂ i khoáº£n khĂ´ng tá»“n táº¡i.");
  }

  const summary = await buildCheckoutSummary(userId, payload);
  const orderNumber = await generateOrderNumber();

  const createdOrder = await Order.create({
    orderNumber,
    userId: new Types.ObjectId(userId),
    couponId: summary.coupon ? new Types.ObjectId(summary.coupon.id) : null,
    customerName: summary.shippingAddress.customerName,
    customerEmail: user.email,
    customerPhone: summary.shippingAddress.customerPhone,
    shippingAddress: summary.shippingAddress.addressLine,
    shippingWard: summary.shippingAddress.ward,
    shippingDistrict: summary.shippingAddress.district,
    shippingProvince: summary.shippingAddress.province,
    subtotal: summary.subtotal,
    discountAmount: summary.discountAmount,
    shippingFee: summary.shippingFee,
    taxAmount: summary.taxAmount,
    totalAmount: summary.totalAmount,
    paymentMethod: payload.paymentMethod,
    paymentStatus: "pending",
    status: "pending",
    note: payload.note ?? null,
  });

  const createdItems = await OrderItem.insertMany(
    summary.items.map((item) => ({
      orderId: createdOrder._id,
      productId: new Types.ObjectId(item.productId),
      variantId: item.variantId ? new Types.ObjectId(item.variantId) : null,
      productName: item.productName,
      productSku: item.productSku,
      variantSku: item.variantSku,
      quantity: item.quantity,
      unitPrice: item.unitPrice,
      discountAmount: 0,
      totalPrice: item.totalPrice,
    }))
  );

  for (const item of summary.items) {
    if (item.variantId) {
      const decremented = await ProductVariant.updateOne(
        { _id: new Types.ObjectId(item.variantId), stockQuantity: { $gte: item.quantity } },
        { $inc: { stockQuantity: -item.quantity } }
      );
      if (decremented.modifiedCount === 0) {
        throw new AppError(
          400,
          `Sáº£n pháº©m ${item.productName} khĂ´ng Ä‘á»§ tá»“n kho (yĂªu cáº§u ${item.quantity}).`
        );
      }
    }
  }

  if (summary.coupon) {
    await Coupon.updateOne(
      { _id: new Types.ObjectId(summary.coupon.id) },
      { $inc: { usedCount: 1 } }
    );
    await CouponUsage.create({
      couponId: new Types.ObjectId(summary.coupon.id),
      userId: new Types.ObjectId(userId),
      orderId: createdOrder._id,
      discountAmount: summary.discountAmount,
    });
  }

  await Cart.deleteMany({
    userId: new Types.ObjectId(userId),
    _id: { $in: summary.items.map((item) => new Types.ObjectId(item.cartItemId)) },
  });

  return { order: createdOrder, items: createdItems };
};

export const updateOrderStatusAdmin = async (
  orderId: string,
  nextStatus: OrderStatus
): Promise<IOrder> => {
  const order = await Order.findById(new Types.ObjectId(orderId));

  if (!order) {
    throw createHttpError(404, "ÄÆ¡n hĂ ng khĂ´ng tá»“n táº¡i.");
  }

  const allowedTargets = ORDER_STATUS_TRANSITIONS[order.status];

  if (!allowedTargets.includes(nextStatus)) {
    throw createHttpError(
      400,
      `KhĂ´ng thá»ƒ chuyá»ƒn Ä‘Æ¡n hĂ ng tá»« tráº¡ng thĂ¡i '${order.status}' sang '${nextStatus}'.`
    );
  }

  if (order.status === "pending" && nextStatus === "confirmed") {
    order.confirmedAt = new Date();
  }
  if (nextStatus === "shipping") {
    order.shippedAt = new Date();
  }
  if (nextStatus === "delivered") {
    order.deliveredAt = new Date();
  }
  if (nextStatus === "completed") {
    order.completedAt = new Date();
  }

  order.status = nextStatus;

  // COD: ghi nháº­n tiá»n ngay khi giao hĂ ng (delivered) â€” "completed" chá»‰ lĂ 
  // khĂ¡ch xĂ¡c nháº­n khĂ´ng cĂ³ váº¥n Ä‘á». Giá»¯ nhĂ¡nh completed cho Ä‘Æ°á»ng rĂºt ngáº¯n
  // shipping -> completed; recordCodPayment tá»± chá»‘ng ghi trĂ¹ng (idempotent).
  if (
    (nextStatus === "delivered" || nextStatus === "completed") &&
    order.paymentMethod === "COD"
  ) {
    await recordCodPayment(order);
  }

  if (nextStatus === "cancelled") {
    order.cancelledAt = new Date();
    await order.save();
    await restockOrderItems(order._id);
    await rollbackCouponUsage(order);
    // Admin há»§y Ä‘Æ¡n Ä‘Ă£ thanh toĂ¡n â†’ tá»± hoĂ n tiá»n Ä‘áº§y Ä‘á»§ (Payment + Order).
    if (order.paymentStatus === "paid") {
      order.paymentStatus = "refunded";
      await order.save();
      await refundPaidPaymentForOrder(order._id, "ÄÆ¡n hĂ ng bá»‹ há»§y bá»Ÿi quáº£n trá»‹ viĂªn.");
    }
    return order;
  }

  if (nextStatus === "returned") {
    await order.save();
    await restockOrderItems(order._id);

const formatVnpDate = (date: Date): string => {
  const pad = (value: number): string => String(value).padStart(2, "0");
  return [
    date.getFullYear(),
    pad(date.getMonth() + 1),
    pad(date.getDate()),
    pad(date.getHours()),
    pad(date.getMinutes()),
    pad(date.getSeconds()),
  ].join("");
};

export const buildVnpaySignature = (
  params: Record<string, string>,
  hashSecret: string
): string => {
  const sortedKeys = Object.keys(params)
    .filter(
      (key) =>
        key !== "vnp_SecureHash" &&
        params[key] !== undefined &&
        params[key] !== null &&
        params[key] !== ""
    )
    .sort();
  const signData = sortedKeys.map((key) => `${key}=${params[key]}`).join("&");
  return crypto
    .createHmac("sha512", hashSecret)
    .update(signData)
    .digest("hex")
    .toUpperCase();
};

export const verifyVnpayCallback = (
  query: Record<string, string>
): boolean => {
  const receivedHash = query.vnp_SecureHash;
  if (!receivedHash) {
    return false;
  }
  return (
    buildVnpaySignature(query, VNPAY_CONFIG.hashSecret) ===
    receivedHash.toUpperCase()
  );
};

export const verifyMomoCallback = (
  body: Record<string, string>
): boolean => {
  const receivedSignature = body.signature;
  if (!receivedSignature) {
    return false;
  }
  const dataToSign = Object.keys(body)
    .filter((key) => key !== "signature" && body[key] !== "" && body[key] !== undefined)
    .sort()
    .map((key) => `${key}=${body[key]}`)
    .join("&");
  const expected = crypto
    .createHmac("sha256", MOMO_CONFIG.secretKey)
    .update(dataToSign)
    .digest("hex");
  return expected === receivedSignature;
};

export const verifyZalopayCallback = (
  data: string,
  mac: string
): boolean => {
  if (!data || !mac) {
    return false;
  }
  const expected = crypto
    .createHmac("sha256", ZALOPAY_CONFIG.key2)
    .update(data)
    .digest("hex");
  return expected === mac;
};

const buildVnpayPaymentUrl = (
  order: IOrder
): { paymentUrl: string; expiredAt: Date } => {
  const createdAt = new Date();
  const expiredAt = new Date(createdAt.getTime() + SESSION_DURATION_MS);
  const params: Record<string, string> = {
    vnp_Version: VNPAY_CONFIG.version,
    vnp_Command: "pay",
    vnp_TmnCode: VNPAY_CONFIG.tmnCode,
    vnp_Amount: String(order.totalAmount * 100),

    vnp_CreateDate: formatVnpDate(createdAt),
    vnp_ExpireDate: formatVnpDate(expiredAt),
  };
  const secureHash = buildVnpaySignature(params, VNPAY_CONFIG.hashSecret);
  const queryString = new URLSearchParams(params).toString();
  return {
    paymentUrl: `${VNPAY_CONFIG.payUrl}?${queryString}&vnp_SecureHash=${secureHash}`,
    expiredAt,
  };
};

export const createPayment = async (
  userId: string,
  orderId: string
): Promise<{
  payment: IPayment;
  paymentUrl: string | null;
  bankTransferInfo: ReturnType<typeof buildBankTransferInfo> | null;
}> => {
  const order = await Order.findOne({
    _id: new Types.ObjectId(orderId),
    userId: new Types.ObjectId(userId),
  });

  if (!order) {
    throw createHttpError(404, "ÄÆ¡n hĂ ng khĂ´ng tá»“n táº¡i hoáº·c khĂ´ng thuá»™c vá» báº¡n.");
  }

  if (!["pending", "confirmed"].includes(order.status)) {

  }

  }

  const activePayment = await Payment.findOne({
    orderId: order._id,
    status: { $in: ["pending"] },
  }).sort({ createdAt: -1 });

  if (
    activePayment?.paymentUrl &&
    activePayment.expiredAt &&
    activePayment.expiredAt.getTime() > Date.now()
  ) {
    return {
      payment: activePayment,
      paymentUrl: activePayment.paymentUrl,
      bankTransferInfo:
        order.paymentMethod === "BANK" ? buildBankTransferInfo(order) : null,
    };
  }

  if (activePayment) {
    activePayment.status = "failed";
    activePayment.failedReason = "PhiĂªn thanh toĂ¡n háº¿t háº¡n.";
    await activePayment.save();
  }

  let paymentUrl: string | null = null;
  let expiredAt: Date | null = null;
  let gatewayOrderId: string | null = null;

  if ((ONLINE_METHODS as readonly string[]).includes(order.paymentMethod)) {
    if (PAYMENT_ENV === "mock") {
      const session = buildMockPaymentUrl(order);
      paymentUrl = session.paymentUrl;
      expiredAt = session.expiredAt;
    } else if (order.paymentMethod === "VNPAY") {
      const session = buildVnpayPaymentUrl(order);
      paymentUrl = session.paymentUrl;
      expiredAt = session.expiredAt;
    } else if (order.paymentMethod === "MOMO") {
      gatewayOrderId = `${order.orderNumber}-${Date.now()}`;
    } else if (order.paymentMethod === "ZALOPAY") {
      gatewayOrderId = `${formatVnpDate(new Date()).slice(4)}_${order.orderNumber}`;
    }
  }

  const payment = await Payment.create({
    orderId: order._id,
    paymentMethod: order.paymentMethod,
    amount: order.totalAmount,
    currency: "VND",
    orderInfo: `Thanh toan don hang ${order.orderNumber}`,
    paymentUrl,
    expiredAt,
    gatewayOrderId,
    status: "pending",
  });

  return {
    payment,
    paymentUrl,
    bankTransferInfo:
      order.paymentMethod === "BANK" ? buildBankTransferInfo(order) : null,
  };
};

export const applyGatewayResult = async (
  input: GatewayResultInput
): Promise<{ alreadyProcessed: boolean; payment: IPayment }> => {
  const order = await Order.findOne({ orderNumber: input.orderNumber });

  if (!order) {
    throw createHttpError(404, `KhĂ´ng tĂ¬m tháº¥y Ä‘Æ¡n hĂ ng ${input.orderNumber}.`);
  }

  const existingByTransaction = await Payment.findOne({
    transactionCode: input.transactionCode,
  });

  if (existingByTransaction && existingByTransaction.status === "paid") {
    return { alreadyProcessed: true, payment: existingByTransaction };
  }

  if (input.amount !== order.totalAmount) {
    throw createHttpError(
      400,
      "Sá»‘ tiá»n giao dá»‹ch khĂ´ng khá»›p vá»›i giĂ¡ trá»‹ Ä‘Æ¡n hĂ ng."
    );
  }

  let payment =
    existingByTransaction ??
    (await Payment.findOne({
      orderId: order._id,
      status: { $in: ["pending"] },
    }).sort({ createdAt: -1 }));

  if (!payment) {
    payment = new Payment({
      orderId: order._id,
      paymentMethod: input.gateway,
      amount: order.totalAmount,
      currency: "VND",
      status: "pending",
    });
  }

  payment.transactionCode = input.transactionCode;
  payment.gatewayResponse = JSON.stringify(input.rawPayload);

  if (input.success) {
    payment.status = "paid";
    payment.paidAt = new Date();
    payment.failedReason = null;
    order.paymentStatus = "paid";
  } else {
    payment.status = "failed";
    payment.failedReason = input.message || "Giao dá»‹ch khĂ´ng thĂ nh cĂ´ng.";
    if (order.paymentStatus !== "paid") {
      order.paymentStatus = "failed";
    }
  }

  await Promise.all([payment.save(), order.save()]);

  return { alreadyProcessed: false, payment };
};
